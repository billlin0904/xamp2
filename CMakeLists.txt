cmake_minimum_required(VERSION 3.24)

project(xamp
    VERSION 1.0.2
    LANGUAGES CXX)

option(XAMP_BUILD_ARCHITECTURE_TESTS "Build playback queue/state regression tests" OFF)
if(XAMP_BUILD_ARCHITECTURE_TESTS)
    enable_testing()
endif()


option(XAMP_BUILD_OUTPUT_DEVICE "Build the cross-platform audio output device library." ON)
option(XAMP_BUILD_STREAM "Build the cross-platform stream library." ON)
option(XAMP_BUILD_PLAYER "Build the cross-platform player library." ON)
option(XAMP_BUILD_METADATA "Build the cross-platform metadata library." ON)
option(XAMP_BUILD_WIDGET_SHARED "Build the Qt widget_shared library on Linux." ON)
option(XAMP_BUILD_APP "Build the Qt xamp executable on Linux." ON)
option(XAMP_BUILD_BENCH "Build the xamp benchmark executable." ON)

if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
    if(XAMP_BUILD_OUTPUT_DEVICE)
        add_subdirectory(src/xamp_base)
        add_subdirectory(src/xamp_output_device)
    endif()

    if(XAMP_BUILD_STREAM)
        if(NOT TARGET xamp_base)
            add_subdirectory(src/xamp_base)
        endif()
        add_subdirectory(src/xamp_stream)
    endif()

    if(XAMP_BUILD_PLAYER)
        if(NOT TARGET xamp_base)
            add_subdirectory(src/xamp_base)
        endif()
        if(NOT TARGET xamp_output_device)
            add_subdirectory(src/xamp_output_device)
        endif()
        if(NOT TARGET xamp_stream)
            add_subdirectory(src/xamp_stream)
        endif()
        add_subdirectory(src/xamp_player)
    endif()

    if(XAMP_BUILD_METADATA)
        if(NOT TARGET xamp_base)
            add_subdirectory(src/xamp_base)
        endif()
        add_subdirectory(src/xamp_metadata)
    endif()

    if(XAMP_BUILD_WIDGET_SHARED)
        if(NOT TARGET xamp_base)
            add_subdirectory(src/xamp_base)
        endif()

        set(QCORO_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
        set(QCORO_BUILD_TESTING OFF CACHE BOOL "" FORCE)
        set(QCORO_WITH_QTDBUS OFF CACHE BOOL "" FORCE)
        set(QCORO_WITH_QTWEBSOCKETS OFF CACHE BOOL "" FORCE)
        set(QCORO_WITH_QTQUICK OFF CACHE BOOL "" FORCE)
        set(QCORO_WITH_QML OFF CACHE BOOL "" FORCE)
        set(QCORO_WITH_QTTEST OFF CACHE BOOL "" FORCE)
        add_subdirectory(src/thirdparty/qcoro)

        set(QWINDOWKIT_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
        set(QWINDOWKIT_BUILD_DOCUMENTATIONS OFF CACHE BOOL "" FORCE)
        set(QWINDOWKIT_BUILD_QUICK OFF CACHE BOOL "" FORCE)
        set(QWINDOWKIT_INSTALL OFF CACHE BOOL "" FORCE)
        add_subdirectory(src/thirdparty/qwindowkit)

        add_subdirectory(src/widget_shared)
    endif()

    if(XAMP_BUILD_APP)
        if(NOT TARGET xamp_base)
            add_subdirectory(src/xamp_base)
        endif()
        if(NOT TARGET xamp_output_device)
            add_subdirectory(src/xamp_output_device)
        endif()
        if(NOT TARGET xamp_stream)
            add_subdirectory(src/xamp_stream)
        endif()
        if(NOT TARGET xamp_player)
            add_subdirectory(src/xamp_player)
        endif()
        if(NOT TARGET xamp_metadata)
            add_subdirectory(src/xamp_metadata)
        endif()
        if(NOT TARGET widget_shared)
            set(QCORO_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
            set(QCORO_BUILD_TESTING OFF CACHE BOOL "" FORCE)
            set(QCORO_WITH_QTDBUS OFF CACHE BOOL "" FORCE)
            set(QCORO_WITH_QTWEBSOCKETS OFF CACHE BOOL "" FORCE)
            set(QCORO_WITH_QTQUICK OFF CACHE BOOL "" FORCE)
            set(QCORO_WITH_QML OFF CACHE BOOL "" FORCE)
            set(QCORO_WITH_QTTEST OFF CACHE BOOL "" FORCE)
            add_subdirectory(src/thirdparty/qcoro)

            set(QWINDOWKIT_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
            set(QWINDOWKIT_BUILD_DOCUMENTATIONS OFF CACHE BOOL "" FORCE)
            set(QWINDOWKIT_BUILD_QUICK OFF CACHE BOOL "" FORCE)
            set(QWINDOWKIT_INSTALL OFF CACHE BOOL "" FORCE)
            add_subdirectory(src/thirdparty/qwindowkit)

            add_subdirectory(src/widget_shared)
        endif()
        add_subdirectory(src/xamp)
    endif()

    if(XAMP_BUILD_BENCH)
        if(NOT TARGET xamp_base)
            add_subdirectory(src/xamp_base)
        endif()
        add_subdirectory(src/bench)
    endif()
else()
    message(STATUS "No root-level targets enabled for ${CMAKE_SYSTEM_NAME}.")
endif()

option(XAMP_BUILD_TESTS "Build Catch2 audio regression tests" OFF)
if(XAMP_BUILD_TESTS)
    enable_testing()
    add_subdirectory(tests)
endif()
