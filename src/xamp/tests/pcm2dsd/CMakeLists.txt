cmake_minimum_required(VERSION 3.24)
project(pcm2dsd_integration_tests LANGUAGES CXX)
enable_testing()
find_package(Qt6 REQUIRED COMPONENTS Core Gui Widgets)
get_filename_component(REPO "${CMAKE_CURRENT_LIST_DIR}/../../../.." ABSOLUTE)
set(XAMP_RUNTIME "${REPO}/src/xamp/x64/Release" CACHE PATH "Built Release runtime")
add_executable(pcm2dsd_stream_test stream_test.cpp)
target_compile_features(pcm2dsd_stream_test PRIVATE cxx_std_23)
target_compile_definitions(pcm2dsd_stream_test PRIVATE UNICODE _UNICODE _ENABLE_EXTENDED_ALIGNED_STORAGE)
target_compile_options(pcm2dsd_stream_test PRIVATE /utf-8 /Zc:__cplusplus)
target_include_directories(pcm2dsd_stream_test PRIVATE "${REPO}/src/xamp_output_device/include")
foreach(module xamp_base xamp_stream)
    target_include_directories(pcm2dsd_stream_test PRIVATE "${REPO}/src/${module}/include")
    target_link_libraries(pcm2dsd_stream_test PRIVATE "${XAMP_RUNTIME}/${module}.lib")
endforeach()
foreach(dep spdlog mimalloc unordered_dense)
    target_include_directories(pcm2dsd_stream_test PRIVATE "${REPO}/src/thirdparty/${dep}/include")
endforeach()
target_include_directories(pcm2dsd_stream_test PRIVATE "${REPO}/src/thirdparty/pcm_dsd_converter/include")
target_link_libraries(pcm2dsd_stream_test PRIVATE Qt6::Core "${REPO}/src/thirdparty/pcm_dsd_converter/dist/lib/pcm_dsd_converter.lib")
set_target_properties(pcm2dsd_stream_test PROPERTIES RUNTIME_OUTPUT_DIRECTORY_RELEASE "${XAMP_RUNTIME}")
add_test(NAME pcm2dsd_stream COMMAND pcm2dsd_stream_test)
set_tests_properties(pcm2dsd_stream PROPERTIES WORKING_DIRECTORY "${XAMP_RUNTIME}")
add_executable(pcm2dsd_settings_test settings_test.cpp)
target_compile_features(pcm2dsd_settings_test PRIVATE cxx_std_23)
target_compile_definitions(pcm2dsd_settings_test PRIVATE UNICODE _UNICODE _ENABLE_EXTENDED_ALIGNED_STORAGE)
target_compile_options(pcm2dsd_settings_test PRIVATE /utf-8 /Zc:__cplusplus)
foreach(module xamp_base xamp_stream xamp_player xamp_output_device xamp_metadata)
    target_include_directories(pcm2dsd_settings_test PRIVATE "${REPO}/src/${module}/include")
endforeach()
foreach(dep spdlog mimalloc unordered_dense)
    target_include_directories(pcm2dsd_settings_test PRIVATE "${REPO}/src/thirdparty/${dep}/include")
endforeach()
target_include_directories(pcm2dsd_settings_test PRIVATE "${REPO}/src/widget_shared")
target_link_libraries(pcm2dsd_settings_test PRIVATE Qt6::Core Qt6::Gui Qt6::Widgets
    "${XAMP_RUNTIME}/widget_shared.lib" "${XAMP_RUNTIME}/xamp_base.lib")
set_target_properties(pcm2dsd_settings_test PROPERTIES RUNTIME_OUTPUT_DIRECTORY_RELEASE "${XAMP_RUNTIME}")
add_test(NAME pcm2dsd_settings COMMAND pcm2dsd_settings_test -platform offscreen)
set_tests_properties(pcm2dsd_settings PROPERTIES WORKING_DIRECTORY "${XAMP_RUNTIME}")
