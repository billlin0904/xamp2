cmake_minimum_required(VERSION 3.24)
project(database_regression LANGUAGES CXX)
enable_testing()
find_package(Qt6 REQUIRED COMPONENTS Core Gui Widgets Sql)
get_filename_component(REPO "${CMAKE_CURRENT_LIST_DIR}/../../../.." ABSOLUTE)
set(XAMP_RUNTIME "${REPO}/src/xamp/x64/Release" CACHE PATH "Built Release runtime")
set(migrations 2024-01-12-093210_init 2026-05-22-110000_scan_insert_indexes 2026-06-15-120000_playlist_album_states)
set(resources "<RCC><qresource prefix=\"/xamp\">")
foreach(migration IN LISTS migrations)
  string(APPEND resources "<file alias=\"migrations/${migration}/up.sql\">${REPO}/src/xamp/migrations/${migration}/up.sql</file>")
endforeach()
string(APPEND resources "</qresource></RCC>")
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/migrations.qrc" "${resources}")
add_executable(music_database_test music_database_test.cpp "${CMAKE_CURRENT_BINARY_DIR}/migrations.qrc")
set_target_properties(music_database_test PROPERTIES AUTORCC ON RUNTIME_OUTPUT_DIRECTORY_RELEASE "${XAMP_RUNTIME}")
target_compile_features(music_database_test PRIVATE cxx_std_23)
target_compile_options(music_database_test PRIVATE /utf-8 /Zc:__cplusplus)
target_compile_definitions(music_database_test PRIVATE UNICODE _UNICODE _ENABLE_EXTENDED_ALIGNED_STORAGE)
target_include_directories(music_database_test PRIVATE "${REPO}/src/widget_shared")
foreach(module xamp_base xamp_stream xamp_player xamp_output_device xamp_metadata)
  target_include_directories(music_database_test PRIVATE "${REPO}/src/${module}/include")
endforeach()
foreach(dep spdlog mimalloc unordered_dense)
  target_include_directories(music_database_test PRIVATE "${REPO}/src/thirdparty/${dep}/include")
endforeach()
target_link_libraries(music_database_test PRIVATE Qt6::Core Qt6::Gui Qt6::Widgets Qt6::Sql
  "${XAMP_RUNTIME}/widget_shared.lib" "${XAMP_RUNTIME}/xamp_base.lib")
add_test(NAME music_database COMMAND music_database_test)
set_tests_properties(music_database PROPERTIES WORKING_DIRECTORY "${XAMP_RUNTIME}")
