find_package(Qt6 REQUIRED COMPONENTS Widgets Test)
if(NOT TARGET widget_shared OR NOT TARGET xamp)
    message(FATAL_ERROR "Shortcut tests require the widget_shared and xamp targets")
endif()
set(CMAKE_AUTOMOC ON)
add_executable(globalshortcut_test globalshortcut_test.cpp)
target_compile_features(globalshortcut_test PRIVATE cxx_std_23)
target_include_directories(globalshortcut_test PRIVATE $<TARGET_PROPERTY:xamp,INCLUDE_DIRECTORIES>)
target_link_libraries(globalshortcut_test PRIVATE widget_shared Qt6::Test)
add_test(NAME globalshortcut_test COMMAND globalshortcut_test)
set_tests_properties(globalshortcut_test PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
